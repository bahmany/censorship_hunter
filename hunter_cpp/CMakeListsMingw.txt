# ─── Software-OpenGL fallback (Windows-only) ───
# Some real Windows machines have no working hardware OpenGL driver (VMs with
# a 2D-only display adapter, some RDP sessions, old GPUs) — GLFW fails outright
# and the GUI can never open. When ON, embeds Mesa's opengl32.dll +
# libgallium_wgl.dll (see tools/embed_gl_fallback.py); the app extracts them
# next to itself and relaunches with software rendering ONLY if hardware
# OpenGL actually fails — hardware-capable machines are completely unaffected.
option(HUNTER_EMBED_GL_FALLBACK "Embed a software-OpenGL fallback (Mesa) for machines with no hardware OpenGL" OFF)
set(HUNTER_GL_FALLBACK_BIN_DIR "${CMAKE_SOURCE_DIR}/../bin/windows/gl_fallback" CACHE PATH "Directory containing opengl32.dll + libgallium_wgl.dll to embed")
set(HUNTER_GL_FALLBACK_OUT_DIR "${CMAKE_BINARY_DIR}/embedded_gl_fallback" CACHE PATH "Output dir for generated GL fallback embed files")

if(HUNTER_EMBED_GL_FALLBACK)
    add_custom_command(
        OUTPUT
            ${HUNTER_GL_FALLBACK_OUT_DIR}/gl_fallback_embedded.h
            ${HUNTER_GL_FALLBACK_OUT_DIR}/gl_fallback_embedded.S
            ${HUNTER_GL_FALLBACK_OUT_DIR}/gl_fallback_meta.json
        COMMAND ${CMAKE_COMMAND} -E make_directory ${HUNTER_GL_FALLBACK_OUT_DIR}
        COMMAND python3 ${CMAKE_SOURCE_DIR}/tools/embed_gl_fallback.py
                --bin ${HUNTER_GL_FALLBACK_BIN_DIR}
                --outdir ${HUNTER_GL_FALLBACK_OUT_DIR}
        DEPENDS ${HUNTER_GL_FALLBACK_BIN_DIR}/opengl32.dll ${HUNTER_GL_FALLBACK_BIN_DIR}/libgallium_wgl.dll
                ${CMAKE_SOURCE_DIR}/tools/embed_gl_fallback.py
        COMMENT "Embedding software-OpenGL fallback (Mesa) via zstd + .incbin"
        VERBATIM
    )
    add_library(gl_fallback_embedded_obj OBJECT ${HUNTER_GL_FALLBACK_OUT_DIR}/gl_fallback_embedded.S)
    target_include_directories(gl_fallback_embedded_obj PRIVATE ${HUNTER_GL_FALLBACK_OUT_DIR})
    set(HUNTER_GL_FALLBACK_INCLUDE_DIR ${HUNTER_GL_FALLBACK_OUT_DIR})
    set(HUNTER_GL_FALLBACK_OBJECT $<TARGET_OBJECTS:gl_fallback_embedded_obj>)
else()
    file(MAKE_DIRECTORY ${HUNTER_GL_FALLBACK_OUT_DIR})
    file(WRITE ${HUNTER_GL_FALLBACK_OUT_DIR}/gl_fallback_embedded.h
"// Stub — GL fallback disabled (HUNTER_EMBED_GL_FALLBACK=OFF).\n\n"
"#pragma once\n"
"#define HUNTER_GL_FALLBACK_EMBEDDED 0\n"
)
    set(HUNTER_GL_FALLBACK_INCLUDE_DIR ${HUNTER_GL_FALLBACK_OUT_DIR})
    set(HUNTER_GL_FALLBACK_OBJECT "")
endif()

# Static linking
# Subsystem/OS version bumped from the MinGW default (5.2, Windows XP x64 /
# Server 2003 era) to 6.0 (Vista+) — an anomalously old target version is
# itself a minor static-anomaly signal, and 6.0 is a safe floor for Windows
# 11 while still being permissive.
target_link_options(hunter PRIVATE
    -static
    -static-libgcc
    -static-libstdc++
    -mwindows
    -Wl,--major-subsystem-version,6
    -Wl,--minor-subsystem-version,0
    -Wl,--major-os-version,6
    -Wl,--minor-os-version,0
)

set(_hunter_deploy_dir "${CMAKE_SOURCE_DIR}/../bin/windows")

add_custom_command(TARGET hunter POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${_hunter_deploy_dir}"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:hunter>" "${_hunter_deploy_dir}/hunter.exe"
)
